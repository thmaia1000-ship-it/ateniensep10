SELECT  
                tbt.ID_turma as 'ID',
                tbt.turma as 'Turma',
                tbt.max_alunos as 'Máximo Alunos',
                (SELECT
                      count(ID_aluno)
                 FROM
                    tb_alunos as tba
                WHERE
                    tba.idturma = tbt.ID_turma and status_turma = 'A'
                )as 'Quantidade Alunos'
            FROM 
            tb_turmas as tbt
    
            ORDER BY turma
